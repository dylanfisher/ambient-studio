# Literals scripts/dupes.js accepts in more than one file, one entry per line:
#   literal "<text>" | files <file> <file>… | <reason>
# Add one only when the files mean different things by the same text: a "utf8" encoding, an
# HTTP method, a flag two programs each define. A fact both files share gets one definition
# and an import instead. The script prints each finding's key, everything before the reason.
literal "data" | files src/server/write-flac.js src/server/write-wav.js | a stream event name in write-flac, the RIFF chunk id in write-wav
literal "number" | files src/engine/descriptor.js src/engine/trajectory.js src/engine/tuning.js src/engine/validate-genome.js src/engine/validation-error.js src/server/export-render.js | the JavaScript typeof tag in each file; descriptor.js also names its number kind with it
literal "string" | files src/engine/descriptor.js src/engine/render.js src/engine/trajectory.js src/engine/tuning.js src/engine/validate-genome.js src/server/export-render.js src/server/render-cli.js | the JavaScript typeof tag; in render-cli, the option type node:util parseArgs takes
literal "${key}" | files src/engine/schema-text.js src/engine/tuning.js | quotes a voice key in the schema text and an unknown tuning-file key in an error
literal "${name}" | files src/engine/tuning.js src/engine/voice-registry.js | quotes a scale name in a tuning error and a parameter name in a voice error
literal "linear" | files src/engine/descriptor.js src/engine/trajectory.js | a parameter scale in descriptor.js, a trajectory segment curve in trajectory.js
literal "${id}" | files src/engine/render.js src/engine/tuning-registry.js | quotes a voice id in a render error and a tuning id in a registry error
literal "lowpass" | files src/engine/drone.js src/engine/noise-bed.js | the Web Audio filter type each voice sets on its own filter
literal "density" | files src/engine/bell.js src/engine/granular-cloud.js | each voice's own parameter: bell notes a minute, grains a second
literal "tone" | files src/engine/noise-bed.js src/engine/pad.js | each voice's own parameter in its macro rows: the pad's harmonic tilt in dB an octave, the noise bed's filter shift in octaves
literal "depth" | files src/engine/noise-bed.js src/engine/pad.js | the noise bed's sweep-and-swell parameter, and a description word in the pad's swell descriptor
literal "utf8" | files src/server/export-render.js src/server/render-cli.js src/server/renderer-versions.js src/server/write-flac.js | the text encoding passed to a read or a stream
literal "boolean" | files src/engine/descriptor.js src/server/render-cli.js | the JavaScript typeof tag in descriptor.js, the option type node:util parseArgs takes in render-cli
literal "duration" | files src/engine/excerpt-offset.js src/server/render-cli.js | a genome parameter named in an excerpt error, and the CLI flag named in a usage error
literal "seed" | files src/engine/prng.js src/server/render-cli.js | the PRNG key named in its range error, and the CLI flag named in a usage error
